module.exports=Object.freeze({name:'security',async run({input={}}) {
  const validAreas=input.affectedAreas===undefined || Array.isArray(input.affectedAreas);
  const areas=Array.isArray(input.affectedAreas)
    ? input.affectedAreas.map(function (area) { return String(area).trim().toLowerCase(); })
    : [];
  const text=[input.title,input.description,input.content].filter(function(value){return typeof value==='string';}).join('\n').slice(0,6000);
  const injectionPatterns=[
    /\b(ignore|disregard)\s+(all\s+)?(previous|prior|above)\s+instructions\b/i,
    /\b(reveal|print|expose|exfiltrate)\s+(the\s+)?(system prompt|secret|api key|token)\b/i,
    /\b(disable|turn off)\s+(rls|row level security|authentication|authorization)\b/i,
    /\b(bypass|override)\s+(auth|authorization|approval|rls)\b/i,
    /تجاهل\s+(كل\s+)?(التعليمات|التوجيهات)\s+(السابقة|أعلاه)/u,
    /اكشف\s+(عن\s+)?(المفتاح|الرمز|التعليمات\s+النظامية)/u,
    /تجاوز\s+(الموافقة|الصلاحيات|الحماية)/u
  ];
  const promptInjection=injectionPatterns.some(function(pattern){return pattern.test(text);});
  const secretExposure=/\b(?:sk-[A-Za-z0-9]{20,}|gh[pousr]_[A-Za-z0-9]{20,})\b/.test(text);
  const blocked=!validAreas || areas.some(x=>['secrets_exposure','rls_disabled','auth_bypass','prompt_injection','data_exfiltration'].includes(x)) || promptInjection || secretExposure;
  const flags={promptInjection,secretExposure};
  return result('security',blocked?'BLOCKED':'PASS',{blocked,reviewRequired:true,areas,flags,tainted:true},blocked?['Untrusted or prohibited input detected']:[]);
}});

'use strict';
const HIGH_RISK_AREAS = new Set(['authentication','authorization','supabase_rls','admin_users','secrets','deployment','github_workflows','telegram_permissions','production_database']);
const TRANSITIONS = Object.freeze({
  PROPOSED:['ANALYZED','REJECTED','FAILED'],
  ANALYZED:['PLANNED','REJECTED','FAILED'],
  PLANNED:['WAITING_APPROVAL','REJECTED','FAILED'],
  WAITING_APPROVAL:['APPROVED','REJECTED'],
  APPROVED:['IMPLEMENTING','REJECTED'],
  IMPLEMENTING:['TESTING','FAILED'],
  TESTING:['VERIFYING','FAILED'],
  VERIFYING:['READY_FOR_REVIEW','FAILED'],
  READY_FOR_REVIEW:['HUMAN_APPROVED','FAILED'],
  HUMAN_APPROVED:['MERGED'],
  MERGED:[], FAILED:[], REJECTED:[]
});
function classifyRisk(areas=[], declared='low') {
  const normalized=Array.isArray(areas)?areas.map(x=>String(x).toLowerCase()):[];
  if (normalized.some(x=>['secrets','deployment','production_database'].includes(x))) return 'critical';
  if (normalized.some(x=>HIGH_RISK_AREAS.has(x))) return 'high';
  if (declared==='critical'||declared==='high') return declared;
  if (declared==='medium') return 'medium';
  return normalized.length ? 'medium' : 'low';
}
function requiresHumanApproval(risk) { return risk==='high'||risk==='critical'; }
function canTransition(from,to,task={}) {
  if (!TRANSITIONS[from] || !TRANSITIONS[from].includes(to)) return false;
  if (task.tainted===true && ['WAITING_APPROVAL','APPROVED','IMPLEMENTING','HUMAN_APPROVED','MERGED'].includes(to)) return false;
  const approvalId=task.approvalId||task.approval_id;
  const humanApprovalId=task.humanApprovalId||task.human_approval_id;
  const humanApproved=task.humanApproved===true||task.human_approved===true;
  if (to==='APPROVED' && !(task.approved===true && approvalId)) return false;
  if (to==='IMPLEMENTING' && !(task.approved===true && approvalId)) return false;
  if (to==='HUMAN_APPROVED' && !(humanApproved && humanApprovalId)) return false;
  if (to==='MERGED' && !(humanApproved && humanApprovalId)) return false;
  return true;
}
module.exports={HIGH_RISK_AREAS,TRANSITIONS,classifyRisk,requiresHumanApproval,canTransition};

'use strict';
const {assess}=require('./risk-engine');
const {route}=require('./router');
const {createTask,transition}=require('./task-store');
const {assertApproval}=require('./approval-gate');
class HaresAISupervisor {
  constructor(agents={}) { this.agents=agents; }
  async analyze(input) {
    const assessment=assess(input);
    // Analysis input is untrusted; trusted tasks must be created through a separate internal path.
    let task=createTask({...input,trustedSource:false},assessment);
    task=transition(task,'ANALYZED');
    const names=route(input.type||'suggestion');
    const reports=[];
    const requiredAgents=new Set(['security']);
    if(input.type==='implementation') requiredAgents.add('verifier');
    const missingRequired=names.find(name=>
      requiredAgents.has(name)&&
      (!this.agents[name]||typeof this.agents[name].run!=='function')
    );
    if(missingRequired){
      const report={
        agent:missingRequired,
        status:'BLOCKED',
        data:{blocked:true,reason:missingRequired==='security'?'Security agent unavailable':'Verifier agent unavailable'},
        warnings:[`Required ${missingRequired} review could not run`]
      };
      return Object.freeze({
        task:{...task,tainted:true,status:'FAILED'},
        reports:[report],
        plan:[report.data]
      });
    }
    for (const name of names) {
      // Developer execution is reserved for implement(), after the approval gate.
      if (name==='developer') continue;
      const agent=this.agents[name];
      if (!agent || typeof agent.run!=='function') {
        continue;
      }
      let report;
      try {
        report=await agent.run({task:{...task},input});
      } catch (error) {
        if (!requiredAgents.has(name)) throw error;
        report={agent:name,status:'BLOCKED',data:{blocked:true,reason:`${name} review failed`},warnings:[`Required ${name} review failed`]};
      }
      reports.push(report);
      if (!report||report.status==='BLOCKED'||report.status==='FAIL'||report.status==='FAILED'||
          (requiredAgents.has(name)&&report.status!=='PASS')) {
        task={...task,tainted:true,status:'FAILED'};
        break;
      }
    }
    return Object.freeze({task,reports,plan:reports.filter(Boolean).map(x=>x.data||{})});
  }
  async implement(task,developer,input) {
    assertApproval(task,'implement');
    if (!developer || typeof developer.run!=='function') throw new Error('Developer agent unavailable');
    return developer.run({task,input});
  }
}
module.exports={HaresAISupervisor};

'use strict';
const {assess}=require('./risk-engine');
const {createAuditLogger}=require('./audit-log');
function createTaskService({env,fetch:fetchImpl=global.fetch}) {
  const audit=createAuditLogger({env,fetch:fetchImpl});
  async function request(path,init={}) {
    if(!env.SUPABASE_URL||!env.SUPABASE_SERVICE_ROLE_KEY) throw new Error('AI task storage is not configured');
    const key=env.SUPABASE_SERVICE_ROLE_KEY;
    const response=await fetchImpl(env.SUPABASE_URL.replace(/\/$/,'')+path,{...init,headers:{apikey:key,Authorization:'Bearer '+key,'Content-Type':'application/json',...(init.headers||{})},signal:AbortSignal.timeout(12000)});
    if(!response.ok) throw new Error('AI task storage request failed');
    return response.status===204?null:response.json();
  }
  return Object.freeze({
    async create(input={}) {
      const risk=assess({affectedAreas:Array.isArray(input.affectedAreas)?input.affectedAreas:[]});
      const rows=await request('/rest/v1/ai_tasks?select=*',{method:'POST',headers:{Prefer:'return=representation'},body:JSON.stringify({
        suggestion_id:input.suggestionId||null,status:'PROPOSED',risk:risk.risk,confidence:null,
        requires_approval:risk.requiresApproval,approved:false,tainted:input.trustedSource!==true,
        source:input.source==='suggestion'?'suggestion':'internal',
        affected_areas:Array.isArray(input.affectedAreas)?input.affectedAreas.filter(x=>typeof x==='string').slice(0,30):[]
      })});
      const task=rows[0];
      try{await audit.record({event:'ai.task.created',suggestionId:task.suggestion_id,metadata:{task_id:task.id,risk:task.risk,affected_areas:task.affected_areas}});}catch{console.error('AI task audit event could not be recorded');}
      return task;
    },
    async get(id) {
      const rows=await request('/rest/v1/ai_tasks?id=eq.'+encodeURIComponent(id)+'&select=*&limit=1');
      return rows[0]||null;
    },
    async recordHumanApproval({taskId,actorUserId,action,decision,reason=''}) {
      if(!taskId||!actorUserId||!['IMPLEMENT','MERGE'].includes(action)||!['APPROVED','REJECTED'].includes(decision)) throw new TypeError('Invalid approval request');
      const rows=await request('/rest/v1/rpc/record_ai_approval',{method:'POST',body:JSON.stringify({p_task_id:taskId,p_actor_user_id:actorUserId,p_action:action,p_decision:decision,p_reason:String(reason).slice(0,1000)})});
      return rows;
    }
  });
}
module.exports={createTaskService};

'use strict';
const {canTransition}=require('./policy');
function transition(task,next,approval={}) {
  if (!task || !canTransition(task.status,next,{...task,...approval})) throw new Error('Invalid or unauthorized AI task transition');
  return {...task,...approval,status:next,updatedAt:new Date().toISOString()};
}

const TYPES = {
  '.aac': { mimeType: 'audio/aac', kind: 'audio', category: 'audio' },
  '.epub': { mimeType: 'application/epub+zip', kind: 'document', category: 'books' },
  '.flac': { mimeType: 'audio/flac', kind: 'audio', category: 'audio' },
  '.jpeg': { mimeType: 'image/jpeg', kind: 'image', category: 'images' },
  '.jpg': { mimeType: 'image/jpeg', kind: 'image', category: 'images' },
  '.m4a': { mimeType: 'audio/mp4', kind: 'audio', category: 'audio' },
  '.m4v': { mimeType: 'video/mp4', kind: 'video', category: 'video' },
  '.mp3': { mimeType: 'audio/mpeg', kind: 'audio', category: 'audio' },
  '.mp4': { mimeType: 'video/mp4', kind: 'video', category: 'video' },
  '.oga': { mimeType: 'audio/ogg', kind: 'audio', category: 'audio' },
  '.ogg': { mimeType: 'audio/ogg', kind: 'audio', category: 'audio' },
  '.opus': { mimeType: 'audio/ogg', kind: 'audio', category: 'audio' },
  '.pdf': { mimeType: 'application/pdf', kind: 'document', category: 'books' },
  '.png': { mimeType: 'image/png', kind: 'image', category: 'images' },
  '.txt': { mimeType: 'text/plain', kind: 'document', category: 'books' },
  '.wav': { mimeType: 'audio/wav', kind: 'audio', category: 'audio' },
  '.webm': { mimeType: 'video/webm', kind: 'video', category: 'video' },
  '.webp': { mimeType: 'image/webp', kind: 'image', category: 'images' }

function categoryFromCaption(caption, fallback) {
  if (/(?:^|\s)#(?:صوتيات|صوت|audio)(?=$|\s)/iu.test(caption)) return 'audio';
  if (/(?:^|\s)#(?:فيديو|مرئي|video)(?=$|\s)/iu.test(caption)) return 'video';
  if (/(?:^|\s)#(?:كتب|كتاب|books?)(?=$|\s)/iu.test(caption)) return 'books';
  if (/(?:^|\s)#(?:صور|image|photos?)(?=$|\s)/iu.test(caption)) return 'images';
  if (/(?:^|\s)#(?:أخرى|اخرى|other)(?=$|\s)/iu.test(caption)) return 'other';
  return fallback;
}

function cleanTitle(caption, fileName) {
  const withoutTags = caption
    .replace(/#[\p{L}\d_]+/gu, '')
    .replace(/\s+/g, ' ')
    .trim();
  const fallback = path.basename(fileName, path.extname(fileName)).replace(/[_-]+/g, ' ').trim();
  return (withoutTags || fallback || 'ملف من المجموعة').slice(0, 120);
}

function getAttachment(message) {
  const file = message.document || message.audio || message.video || message.voice ||
    message.video_note || message.animation ||
    (message.photo && message.photo[message.photo.length - 1]);
  if (!file) return null;

  const hintedExtension = file.file_name ? path.extname(file.file_name).toLowerCase() : '';
  const extension = hintedExtension ||
    (message.voice ? '.ogg' : message.video_note ? '.mp4' : message.photo ? '.jpg' : '') || ({
    'audio/aac': '.aac',
    'audio/flac': '.flac',
    'audio/mp4': '.m4a',
    'audio/mpeg': '.mp3',
    'audio/ogg': '.ogg',
    'audio/wav': '.wav',
    'image/jpeg': '.jpg',
    'image/png': '.png',
    'image/webp': '.webp',
    'video/mp4': '.mp4',
    'video/webm': '.webm'
  }[file.mime_type] || '');
  const type = TYPES[extension];
  if (!type || !file.file_id || !file.file_unique_id) return null;
  const fileName = (file.file_name || ('telegram-file' + extension))
    .replace(/\\/g, '/')
    .split('/')
    .pop()
    .replace(/[\u0000-\u001f\u007f]/g, '')
    .slice(0, 180) || ('telegram-file' + extension);
  return {
    file: file,
    extension: extension,
    fileName: fileName,
    mimeType: type.mimeType,
    kind: type.kind,
    defaultCategory: type.category
  };
}

function createMediaLibrary(options) {
  options = options || {};
  const env = options.env || process.env;
  const fetchImpl = options.fetch || global.fetch;
  const configuredStorageDirectory = typeof options.storageDirectory === 'string' && options.storageDirectory.trim()
    ? options.storageDirectory.trim()
    : (typeof env.STORAGE_DIRECTORY === 'string' && env.STORAGE_DIRECTORY.trim() ? env.STORAGE_DIRECTORY.trim() : '');
  const storageDirectory = path.resolve(configuredStorageDirectory || path.join(__dirname, '..', 'storage'));
  const filesDirectory = path.join(storageDirectory, 'media');
  const indexPath = path.join(storageDirectory, 'media-index.json');
  const storage = resolveStorage(env, options);
  const useRemote = !!storage;
  const botToken = env.TELEGRAM_BOT_TOKEN;
  const mediaGroupId = env.TELEGRAM_MEDIA_GROUP_ID || env.TELEGRAM_CHAT_ID;
  let items = [];
  let offset = null;
  let initialized = false;
  let running = false;
  let pollPromise = null;

  const remoteIndexKey = 'library/_index.json';

  async function initialize() {
    if (initialized) return;
    if (useRemote) {
      const remoteIndex = await storage.get(remoteIndexKey);
      if (remoteIndex) {
        let stored;
        try {
          stored = JSON.parse(remoteIndex.body.toString('utf8'));
        } catch {
          throw new Error('Invalid remote media library index.');
        }
        if (!stored || !Array.isArray(stored.items) ||
            (stored.offset !== null && !Number.isSafeInteger(stored.offset))) {
          throw new Error('Invalid remote media library index.');
        }
        items = stored.items;
        offset = stored.offset;
      } else {
        // Older remote deployments stored media objects without a durable index.
        // Start with an empty index; never infer approval from object presence.
        items = [];
        offset = null;
        await save();
      }
      initialized = true;
      return;
    }
    await fs.mkdir(filesDirectory, { recursive: true });
    try {
      const contents = await fs.readFile(indexPath, 'utf8');
      const stored = JSON.parse(contents);
      if (!stored || !Array.isArray(stored.items) ||
          (stored.offset !== null && !Number.isSafeInteger(stored.offset))) {
        throw new Error('Invalid media library index.');
      }
      items = stored.items;
      offset = stored.offset;
    } catch (error) {
      if (error.code !== 'ENOENT') throw error;
      await save();
    }
    initialized = true;
  }

  async function save() {
    const serialized = JSON.stringify({ offset: offset, items: items });
    if (useRemote) {
      await storage.put(remoteIndexKey, Buffer.from(serialized, 'utf8'), 'application/json', { upsert: true });
      return;
    }
    await fs.mkdir(storageDirectory, { recursive: true });
    const temporaryPath = indexPath + '.' + randomUUID() + '.tmp';
    await fs.writeFile(temporaryPath, serialized, 'utf8');
    await fs.rename(temporaryPath, indexPath);
  }

  async function storeFile(item, data) {
    if (useRemote) {
      const objectKey = buildKey(item.kind, item.id, item.fileName);
      await storage.put(objectKey, data, item.mimeType);
      const head = await storage.head(objectKey);
      if (!head || head.size !== data.length) throw new Error('Supabase object verification failed.');
      item.objectKey = objectKey;
      return objectKey;
    }
    const storedPath = path.join(filesDirectory, item.storedName);
    await fs.writeFile(storedPath, data, { flag: 'wx' });
    return item.storedName;
  }

  async function deleteStoredFile(item) {
    if (useRemote && item.objectKey) await storage.delete(item.objectKey);
    else await fs.rm(path.join(filesDirectory, item.storedName), { force: true });
  }

  async function telegram(method, payload) {
    const response = await fetchImpl('https://api.telegram.org/bot' + botToken + '/' + method, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json; charset=utf-8',
        Accept: 'application/json; charset=utf-8'
      },
      signal: AbortSignal.timeout(method === 'getUpdates' ? 35000 : 15000),
      body: JSON.stringify(payload)
    });
    let result;
    try {
      result = JSON.parse(new TextDecoder('utf-8', { fatal: true }).decode(await response.arrayBuffer()));
    } catch {
      throw new Error('Telegram returned an invalid response.');
    }
    if (!response.ok || !result.ok) {
      throw new Error('Telegram API request failed for ' + method + '.');
    }
    return result.result;
  }

  async function sendReviewMessage(item) {
    const message = 'ملف جديد للمراجعة قبل نشره في المنصة:\n' +
      'العنوان: ' + item.title + '\n' +
      'التصنيف: ' + CATEGORY_NAMES[item.category] + '\n' +
      'النوع: ' + item.fileName;
    await telegram('sendMessage', {
      chat_id: mediaGroupId,
      text: message,
      reply_markup: {
        inline_keyboard: [[
          { text: 'اعتماد ونشر', callback_data: 'media:approve:' + item.id },
          { text: 'رفض', callback_data: 'media:reject:' + item.id }
        ]]
      }
    });
  }

  async function receiveFile(message) {
    const attachment = getAttachment(message);
    if (!attachment) return;
    if (items.filter(function (item) { return item.status === 'approved' || item.status === 'pending'; }).length >= MAX_LIBRARY_ITEMS) {
      await telegram('sendMessage', {
        chat_id: mediaGroupId,
        text: 'لم يُحفظ الملف؛ وصلت المكتبة إلى الحد المسموح. احذف بعض المواد أولًا.'
      });
      return;
    }
    if (attachment.file.file_size > MAX_FILE_BYTES) {
      await telegram('sendMessage', {
        chat_id: mediaGroupId,
        text: 'لم يُحفظ الملف؛ الحد الأقصى لحجم الملف هو 20 ميجابايت.'
      });
      return;
    }
    if (items.some(function (item) {
      return item.status !== 'removed' && item.fileUniqueId === attachment.file.file_unique_id;
    })) {
      return;
    }

    const fileInfo = await telegram('getFile', { file_id: attachment.file.file_id });
    if (!fileInfo.file_path || fileInfo.file_path.split('/').some(function (segment) {
      return !/^[A-Za-z0-9._-]+$/.test(segment) || segment === '.' || segment === '..';
    })) {
      throw new Error('Telegram returned an invalid file path.');
    }
    const downloadUrl = 'https://api.telegram.org/file/bot' + botToken + '/' +
      fileInfo.file_path.split('/').map(function (segment) {
        return encodeURIComponent(segment);
      }).join('/');
    const download = await fetchImpl(downloadUrl, { signal: AbortSignal.timeout(30000) });
    if (!download.ok) throw new Error('Telegram file download failed.');
    const data = Buffer.from(await download.arrayBuffer());
    if (data.length === 0 || data.length > MAX_FILE_BYTES) {
      throw new Error('Telegram file size is outside the allowed limit.');
    }

    const id = randomUUID();
    const storedName = id + attachment.extension;
    const caption = typeof message.caption === 'string' ? message.caption : '';
    const item = {
      id: id,
      storedName: storedName,
      fileName: path.basename(attachment.fileName).slice(0, 180),
      fileUniqueId: attachment.file.file_unique_id,
      mimeType: attachment.mimeType,
      kind: attachment.kind,
      category: categoryFromCaption(caption, attachment.defaultCategory),
      title: cleanTitle(caption, attachment.fileName),
      status: 'pending',
      uploadedAt: new Date((message.date || Date.now() / 1000) * 1000).toISOString()
    };
    if (useRemote) item.objectKey = buildKey(item.kind, item.id, item.fileName);
    await storeFile(item, data);
    items.push(item);
    try {
      await save();
    } catch (error) {
      items = items.filter(function (candidate) { return candidate.id !== id; });
      await deleteStoredFile(item);
      throw error;
    }
    try {
      await sendReviewMessage(item);
    } catch (error) {
      items = items.filter(function (candidate) { return candidate.id !== id; });
      await save();
      await deleteStoredFile(item);
      throw error;
    }
  }

  async function isGroupAdministrator(userId) {
    if (!userId) return false;
    const member = await telegram('getChatMember', {
      chat_id: mediaGroupId,
      user_id: userId
    });
    return member && ['creator', 'administrator'].includes(member.status);
  }

  async function removeApprovedMedia(message) {
    const match = typeof message.text === 'string' &&
      message.text.match(/^\/media-remove(?:@\w+)?\s+([0-9a-f-]{36})\s*$/i);
    if (!match) return false;
    if (!await isGroupAdministrator(message.from && message.from.id)) {
      await telegram('sendMessage', {
        chat_id: mediaGroupId,
        text: 'حذف مواد المكتبة متاح لمشرفي المجموعة فقط.'
      });
      return true;
    }

    const item = items.find(function (candidate) {
      return candidate.id === match[1] && candidate.status === 'approved';
    });
    if (!item) {
      await telegram('sendMessage', {
        chat_id: mediaGroupId,
        text: 'المادة غير موجودة أو سبق حذفها.'
      });
      return true;
    }

    const previousStatus = item.status;
    item.status = 'removed';
    item.removedAt = new Date().toISOString();
    try {
      await save();
    } catch (error) {
      item.status = previousStatus;
      delete item.removedAt;
      throw error;
    }
    await deleteStoredFile(item);
    await telegram('sendMessage', {
      chat_id: mediaGroupId,
      text: 'حُذفت المادة من مكتبة المنصة: ' + item.title
    });
    return true;
  }

  async function processCallback(callback) {
    const match = typeof callback.data === 'string' && callback.data.match(/^media:(approve|reject):([0-9a-f-]{36})$/i);
    const groupId = mediaGroupId;
    if (!match || !callback.message || String(callback.message.chat.id) !== String(groupId)) {
      await telegram('answerCallbackQuery', {
        callback_query_id: callback.id,
        text: 'الطلب غير صالح.'
      });
      return;
    }

    if (!await isGroupAdministrator(callback.from && callback.from.id)) {
      await telegram('answerCallbackQuery', {
        callback_query_id: callback.id,
        text: 'اعتماد الملفات متاح لمشرفي المجموعة فقط.',
        show_alert: true
      });
      return;
    }

    const item = items.find(function (candidate) {
      return candidate.id === match[2] && candidate.status === 'pending';
    });
    if (!item) {
      await telegram('answerCallbackQuery', {
        callback_query_id: callback.id,
        text: 'تمت معالجة الملف أو لم يعد موجودًا.'
      });
      return;
    }

    const approved = match[1] === 'approve';
    item.status = approved ? 'approved' : 'rejected';
    item.reviewedAt = new Date().toISOString();
    try {
      await save();
    } catch (error) {
      item.status = 'pending';
      delete item.reviewedAt;
      throw error;
    }
    if (!approved) {
      await deleteStoredFile(item);
    }
    try {
      await telegram('answerCallbackQuery', {
        callback_query_id: callback.id,
        text: approved ? 'تم اعتماد الملف ونشره.' : 'تم رفض الملف.'
      });
      await telegram('editMessageText', {
        chat_id: groupId,
        message_id: callback.message.message_id,
        text: (approved
          ? 'تم اعتماد ونشر الملف: ' + item.title + '\nللحذف لاحقًا استخدم /media-remove ' + item.id
          : 'تم رفض الملف: ' + item.title)
      });
    } catch (error) {
      console.error('Telegram media review notification failed: ' + error.message);
    }
  }

  async function processUpdate(update) {
    await initialize();
    const message = update?.message;
    if (message) {
      const chatId = message.chat?.id;
      if (chatId != null && String(chatId) === String(mediaGroupId)) {
        if (await removeApprovedMedia(message)) return;
        await receiveFile(message);
        return;
      }
    }
    if (update?.callback_query) await processCallback(update.callback_query);
  }

  async function pollLoop() {
    while (running) {
      try {
        const updates = await telegram('getUpdates', {
          offset: offset === null ? undefined : offset,
          timeout: 25,
          allowed_updates: ['message', 'callback_query']
        });
        for (const update of updates) {
          if (!running) break;
          await processUpdate(update);
          offset = update.update_id + 1;
          await save();
        }
      } catch (error) {
        console.error('Telegram media polling failed: ' + error.message);
        await new Promise(function (resolve) { setTimeout(resolve, 5000); });
      }
    }
  }

  async function start() {
    if (env.TELEGRAM_MEDIA_ENABLED !== 'true') return;
    if (!botToken || !mediaGroupId) {
      throw new Error('Telegram media requires a bot token and a group ID.');
    }
    await initialize();
    running = true;
    pollPromise = pollLoop();
  }

  async function stop() {
    running = false;
    if (pollPromise) await pollPromise;
  }

  async function listApproved() {
    await initialize();
    return items.filter(function (item) {
      return item.status === 'approved';
    }).sort(function (left, right) {
      return right.uploadedAt.localeCompare(left.uploadedAt);
    }).map(function (item) {
      return {
        id: item.id,
        title: item.title,
        category: item.category,
        categoryName: CATEGORY_NAMES[item.category] || CATEGORY_NAMES.other,
        kind: item.kind,
        mimeType: item.mimeType,
        fileName: item.fileName,
        uploadedAt: item.uploadedAt,
        url: '/api/library/files/' + item.id
      };
    });
  }

  async function getApprovedFile(id) {
    await initialize();
    const item = items.find(function (candidate) {
      return candidate.id === id && candidate.status === 'approved';
    });
    if (!item) return null;
    if (useRemote && item.objectKey) {
      const obj = await storage.get(item.objectKey);
      if (!obj) return null;
      return { item: item, data: obj.body };
    }
    if (useRemote) {
      const objectKey = buildKey(item.kind, item.id, item.fileName);
      const obj = await storage.get(objectKey);
      if (!obj) return null;
      return { item: item, data: obj.body };
    }
    const filePath = path.join(filesDirectory, item.storedName);
    return { item: item, data: await fs.readFile(filePath) };
  }

  return {
    getApprovedFile: getApprovedFile,
    listApproved: listApproved,
    processUpdate: processUpdate,
    start: start,
    stop: stop
  };
}

module.exports = {
  CATEGORY_NAMES: CATEGORY_NAMES,
  createMediaLibrary: createMediaLibrary

'use strict';

class MemoryStore {
  constructor() { this.objects = new Map(); this.ops = []; }
  async put(key, buffer, contentType, options) {
    if (!Buffer.isBuffer(buffer)) buffer = Buffer.from(buffer);
    if (this.objects.has(key) && !(options && options.upsert === true)) throw new Error('OBJECT_EXISTS');
    this.objects.set(key, { body: Buffer.from(buffer), contentType: contentType || 'application/octet-stream' });
    this.ops.push('put:' + key);
  }
  async get(key) {
    const e = this.objects.get(key);
    if (!e) return null;
    this.ops.push('get:' + key);
    return { body: Buffer.from(e.body), contentType: e.contentType, size: e.body.length };
  }
  async head(key) {
    const e = this.objects.get(key);
    if (!e) return null;
    this.ops.push('head:' + key);
    return { contentType: e.contentType, size: e.body.length };
  }
  async exists(key) { return this.objects.has(key); }
  async listKeys(prefix) {
    const out = [...this.objects.keys()].filter(k => !prefix || k.startsWith(prefix));
    this.ops.push('list:' + (prefix || ''));
    return out;
  }
  async delete(key) { const had = this.objects.delete(key); if (had) this.ops.push('delete:' + key); return had; }
}

function buildKey(kind, id, filename) {
  const safe = String(filename || 'file').replace(/\\/g, '/').split('/').pop().replace(/[\u0000-\u001f\u007f]/g, '') || 'file';
  const m = { audio: 'audio', video: 'video', books: 'books', images: 'images', other: 'other' };
  const dir = m[kind] || 'other';
  return `library/${dir}/${id}/${safe}`;
}

function SupabaseMediaStorage(opts) {
  const o = opts || {};
  this.bucket = o.bucket || process.env.SUPABASE_BUCKET || 'haress-media';
  this.supabaseUrl = o.supabaseUrl || process.env.SUPABASE_URL;
  this.serviceRoleKey = o.serviceRoleKey || process.env.SUPABASE_SERVICE_ROLE_KEY;
}

async function sbRequest(path, method, opts, supabaseUrl, serviceKey) {
  const o = opts || {};
  const url = supabaseUrl + '/storage/v1' + path;
  const headers = { 'apikey': serviceKey, 'Authorization': 'Bearer ' + serviceKey, ...(o.headers || {}) };
  const init = { method: method || 'GET', headers: headers };
  if (o.body !== undefined) init.body = o.body;
  if (o.raw) {
    const res = await fetch(url, init);
    return res;
  }
  const res = await fetch(url, init);
  const text = await res.text();
  let data; try { data = text ? JSON.parse(text) : null; } catch { data = text; }
  if (!res.ok) {
    const e = new Error('Supabase storage ' + res.status + ' ' + res.statusText);
    e.status = res.status; e.body = data; e.code = data && (data.error || data.code || data.message);
    throw e;
  }
  return data;
}

SupabaseMediaStorage.prototype.put = async function (key, buffer, contentType, options) {
  const u = this.supabaseUrl; const k = this.serviceRoleKey;
  await sbRequest('/object/' + this.bucket + '/' + encodeURIComponent(key).replace(/%2F/g, '/'), 'POST', {
    headers: { 'Content-Type': contentType || 'application/octet-stream', 'x-upsert': options && options.upsert === true ? 'true' : 'false' },
    body: buffer
  }, u, k);
};
SupabaseMediaStorage.prototype.get = async function (key) {
  const u = this.supabaseUrl; const k = this.serviceRoleKey;
  try {
    const res = await sbRequest('/object/' + this.bucket + '/' + encodeURIComponent(key).replace(/%2F/g, '/'), 'GET', { raw: true }, u, k);
    if (res.status === 404) return null;
    if (!res.ok) throw new Error('Supabase storage ' + res.status);
    const buf = Buffer.from(await res.arrayBuffer());
    return { body: buf, contentType: res.headers.get('content-type') || 'application/octet-stream', size: buf.length };
  } catch (e) { if (e.status === 404) return null; throw e; }
};
SupabaseMediaStorage.prototype.head = async function (key) {
  const g = await this.get(key);
  if (!g) return null;
  return { contentType: g.contentType, size: g.size };
};
SupabaseMediaStorage.prototype.exists = async function (key) { return !!(await this.head(key)); };
SupabaseMediaStorage.prototype.listKeys = async function (prefix) {
  const u = this.supabaseUrl; const k = this.serviceRoleKey;
  // list with prefix split (supabase list is by folder)
  const parts = String(prefix || '').split('/').filter(Boolean);
  const folder = parts.slice(0, -1).join('/');
  const search = parts[parts.length - 1] || '';
  const data = await sbRequest('/object/list/' + this.bucket + '/' + (folder ? encodeURIComponent(folder).replace(/%2F/g, '/') : ''), 'POST', {
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ limit: 100, offset: 0, search: search || '' })
  }, u, k);
  if (!Array.isArray(data)) return [];
  return data.map(e => (folder ? folder + '/' : '') + e.name).filter(n => n !== (prefix || ''));
};
SupabaseMediaStorage.prototype.delete = async function (key) {
  const u = this.supabaseUrl; const k = this.serviceRoleKey;
  await sbRequest('/object/' + this.bucket, 'DELETE', {
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ prefixes: [key] })
  }, u, k);
};

function hasSupabaseConfig(env) {
  return !!(env && env.SUPABASE_URL && env.SUPABASE_SERVICE_ROLE_KEY);
}

function resolveStorage(env, options) {
  if (options && options.storage) return options.storage;
  if (hasSupabaseConfig(env)) return new SupabaseMediaStorage({ bucket: env.SUPABASE_STORAGE_BUCKET, supabaseUrl: env.SUPABASE_URL, serviceRoleKey: env.SUPABASE_SERVICE_ROLE_KEY });
  return null;
}

module.exports = { MemoryStore: MemoryStore, SupabaseMediaStorage: SupabaseMediaStorage, hasSupabaseConfig: hasSupabaseConfig, resolveStorage: resolveStorage, buildKey: buildKey };
